namespace = sbscannon

country_event = {
	id = sbscannon.1
	is_triggered_only = yes
	hide_window = yes
	location = event_target:sbs_cannon_system

	trigger = {
		exists = this
		is_scope_type = country
		is_ai = no
		has_technology = tech_sbs_cannon
		any_owned_starbase = { has_starbase_building = sbs_cannon }
		NOT = { has_country_flag = sbs_cannon_target }
	}

	immediate = {
		set_timed_country_flag = { flag = sbs_cannon_target days = 2700 }
		every_owned_starbase = {
			limit = {
				has_starbase_building = sbs_cannon
				solar_system = {
					NOT = {
						has_megastructure = sbs_cannon
					}
				}
			}
			save_global_event_target_as = sbs_cannon_system
			solar_system = {
				spawn_megastructure = {
					type = sbs_cannon
					owner = space_owner
					planet = star
					orbit_distance = 50
					orbit_angle = 270
					init_effect = {
						prevprev = {
							starbase_event = { id = sbscannon.2 }
						}
					}
				}
			}
		}
		owner = {
			country_event = { id = sbscannon.3 days = 180 }
			country_event = { id = sbscannon.7 days = 2700 }
		}
	}
}

country_event = {
	id = sbscannon.3
	title = "sbscannon.3.name"
	desc = "sbscannon.3.desc"
	picture = GFX_evt_sarmat_out
	show_sound = sbs_event_siren
	location = event_target:sbs_cannon_system

	is_triggered_only = yes

	immediate = {
		random_system_within_border = {
			limit = {
				any_system_megastructure = { is_megastructure_type = sbs_cannon }
				any_country_neighbor_to_system = {
					OR = {
						is_ai = yes
						is_country_type = default
						is_country_type = dormant_marauders
					}
					has_communications = root
					NOR = {
						is_at_war_with = root
						root = { is_protective_to = prev }
						root = { is_friendly_to = prev }
						root = { is_loyal_to = prev }
					}
				}
			}
			save_global_event_target_as = sbs_cannon_system
			random_country_neighbor_to_system = {
				limit = {
					OR = {
						is_ai = yes
						is_country_type = default
						is_country_type = dormant_marauders
					}
					has_communications = root
					NOR = {
						is_at_war_with = root
						root = { is_protective_to = prev }
						root = { is_friendly_to = prev }
						root = { is_loyal_to = prev }
					}
				}
				save_global_event_target_as = sbs_cannon_target
			}
			random_neighbor_system = {
				limit = {
					has_hyperlane_to = prev
					exists = space_owner
					space_owner = {
						OR = {
							is_ai = yes
							is_country_type = default
							is_country_type = dormant_marauders
						}
						has_communications = root
						NOR = {
							is_at_war_with = root
							root = { is_protective_to = prev }
							root = { is_friendly_to = prev }
							root = { is_loyal_to = prev }
						}
					}
				}
				set_star_flag = sbs_cannon_system_star
				save_global_event_target_as = sbs_cannon_system_target
			}
		}		
	}

	option = {
		name = BATTLESTATIONS	
		trigger = {
			owner = {
				resource_stockpile_compare = {
					resource = energy
					value >= 2500
				}
			}
		}
		owner = { add_resource = { energy = -2500 } }

		hidden_effect = {
			random_system_within_border = {
				limit = {
					any_system_megastructure = { is_megastructure_type = sbs_cannon }
				}
				random_neighbor_system = {
					limit = {
						has_star_flag = sbs_cannon_system_star
					}
					solar_system = {
						every_fleet_in_system = { destroy_fleet = this }
						sbs_destroy_star_system = yes
					}
				}
			}
			country_event = { id = sbscannon.8 days = 30 }
		}
	}

	option = {
		name = TOOPOOR
		hidden_effect = {
			country_event = { id = sbscannon.8 days = 30 }
		}
	}
}

country_event = {
	id = sbscannon.5
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		exists = from
		from = {
			has_megastructure = sbs_cannon
			NAND = {
				exists = starbase
				starbase = { has_starbase_building = sbs_cannon }
			}
		}
	}
	immediate = {
		from = {
			random_system_megastructure = {
				limit = { is_megastructure_type = sbs_cannon }
				remove_megastructure = this
				create_ambient_object = {
					type = explosion_particle_object
					location = this
					use_3d_location = yes
					duration = 20
					scale = 5
				}
			}
		}
	}
}
