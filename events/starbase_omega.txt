namespace = sbsomega

country_event = {
	id = sbsomega.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		exists = this
		is_scope_type = country
		is_ai = no
		has_technology = tech_sbs_cannon
		any_owned_starbase = { has_starbase_building = sbs_omega }
		NOT = { has_country_flag = sbs_omega_target }
	}

	immediate = {
		set_timed_country_flag = { flag = sbs_omega_target days = 1800 }
		every_owned_starbase = {
			limit = {
				has_starbase_building = sbs_omega
			}
			solar_system = {
				set_star_flag = sbs_omega_star
			}
		}
		owner = {
			country_event = { id = sbsomega.3 days = 90 }
			country_event = { id = sbsomega.4 days = 1800 }
		}
	}
}

country_event = {
	id = sbsomega.3
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_system = {
			limit = { has_star_flag = sbs_omega_star }
			solar_system = {
				every_system_planet = {
					limit = {
						is_star = no
						is_asteroid = no
						habitable_planet = yes
						colonizable_planet = yes
						NOR = {
							is_planet_class = pc_shattered
							is_planet_class = pc_shattered_2
							is_planet_class = pc_shielded

							uninhabitable_regular_planet = yes
						}
					}
					if = {
						limit = { is_colony = no }
						nuke_pre_ftls_toxic_effect = yes
					}
				}
				every_system_planet = {
					limit = {
						is_planet_class = pc_nuked
					}
					nuke_pre_ftls_toxic_effect = yes
				}
				star = {
					create_ambient_object = {
						type = "destroyed_system"
						location = this
					}
					last_created_ambient_object = {
						set_ambient_object_flag = destroyed_system_effect
						set_location = {
							target = prev
							distance = 0
							angle = random
						}
					}	
				}
			}
		}
		random_list = {
			50 = {
				add_opinion_modifier = {
					modifier = opinion_toxified_my_world
					who = from.owner
				}
			}	
			50 = {	
				add_opinion_modifier = {
					modifier = opinion_cracked_a_primitive_world
					who = from.owner
				}
			}
		}
	}

	after = {
		hidden_effect = {
			random_owned_starbase = {
				limit = {
					has_starbase_building = sbs_omega
				}
				remove_starbase_building = { building = sbs_omega }
			}
		}
	}
}
